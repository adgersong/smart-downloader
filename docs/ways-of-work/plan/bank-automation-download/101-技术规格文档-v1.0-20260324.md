```typescript
// AI 生成流程服务
class WorkflowAIService {
  // 从文字描述生成流程
  async generateFromText(description: string): Promise<WorkflowGraph> {
    // 调用 Qwen VL API 解析自然语言
    const response = await fetch('/api/v1/ai/parse-text', {
      method: 'POST',
      body: JSON.stringify({ description }),
    });
    
    const result = await response.json();
    
    // 转换为 React Flow 节点和边
    return this.convertToWorkflow(result);
  }
  
  // 转换为工作流图
  private convertToWorkflow(aiResult: any): WorkflowGraph {
    const nodes = aiResult.actions.map((action, index) => ({
      id: `node-${index}`,
      type: this.mapActionType(action.type),
      position: { x: 250, y: 100 + index * 150 },
      data: {
        label: action.name,
        icon: getIcon(action.type),
        config: action.config,
      },
    }));
    
    const edges = nodes.slice(0, -1).map((node, index) => ({
      id: `edge-${index}`,
      source: node.id,
      target: nodes[index + 1].id,
      type: 'smoothstep',
      animated: true,
    }));
    
    return { nodes, edges };
  }
}

// 用户调整支持
const WorkflowCanvas = () => {
  const [nodes, setNodes] = useNodesState([]);
  const [edges, setEdges] = useEdgesState([]);
  const [isGenerated, setIsGenerated] = useState(false);
  
  // AI 生成后允许用户调整
  const handleAIGenerate = async (description: string) => {
    const workflow = await aiService.generateFromText(description);
    setNodes(workflow.nodes);
    setEdges(workflow.edges);
    setIsGenerated(true);
  };
  
  // 用户拖拽调整位置
  const onNodesChange = useCallback((changes) => {
    setNodes((nds) =>
      applyNodeChanges(nds, changes)
    );
  }, [setNodes]);
  
  // 用户修改节点属性
  const updateNodeConfig = (nodeId: string, config: NodeConfig) => {
    setNodes((nds) =>
      nds.map((node) =>
        node.id === nodeId
          ? { ...node, data: { ...node.data, config } }
          : node
      )
    );
  };
  
  // 添加新节点
  const addNode = (type: NodeType, position: Position) => {
    const newNode = createNode(type, position);
    setNodes((nds) => nds.concat(newNode));
  };
  
  // 删除节点
  const removeNode = (nodeId: string) => {
    setNodes((nds) => nds.filter((n) => n.id !== nodeId));
    setEdges((eds) =>
      eds.filter((e) => e.source !== nodeId && e.target !== nodeId)
    );
  };
  
  return (
    <div>
      <WorkflowCanvas
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onNodeUpdate={updateNodeConfig}
      />
      {isGenerated && (
        <Alert message="AI 生成完成，您可以拖拽调整节点位置或修改属性" type="info" />
      )}
    </div>
  );
};

    """LangGraph 工作流引擎"""
    
    def __init__(self):
        self.graph = self._build_graph()
    
    def _build_graph(self) -> StateGraph:
        """构建状态机图"""
        workflow = StateGraph(WorkflowState)
        
        # 添加节点
        workflow.add_node("login", self._login_node)
        workflow.add_node("navigate", self._navigate_node)
        workflow.add_node("interact", self._interact_node)
        workflow.add_node("download", self._download_node)
        workflow.add_node("verify", self._verify_node)
        workflow.add_node("error_handler", self._error_handler_node)
        
        # 添加边
        workflow.add_edge("login", "navigate")
        workflow.add_edge("navigate", "interact")
        workflow.add_edge("interact", "download")
        workflow.add_edge("download", "verify")
        workflow.add_edge("verify", END)
        
        # 错误处理边
        workflow.add_conditional_edges(
            "interact",
            self._check_error,
            {
                "success": "download",
                "error": "error_handler"
            }
        )
        
        return workflow.compile()
    
    async def execute(self, workflow_yaml: dict, context: dict) -> WorkflowState:
        """执行工作流"""
        initial_state = WorkflowState(
            current_step=0,
            actions=workflow_yaml["actions"],
            results=[],
            errors=[],
            screenshots=[],
            variables=context
        )
        
        result = await self.graph.ainvoke(initial_state)
        return result

    """人类行为模拟器"""

        self.variation_ms = 50
    
    def generate_mouse_trajectory(
        self, 
        start: Dict[str, float], 
        end: Dict[str, float],
        duration_ms: int = None
    ) -> List[Dict[str, float]]:
        """
        生成拟人化鼠标移动轨迹 (贝塞尔曲线)
        
        特性:
        - 非线性路径 (轻微弯曲)
        - 加速度/减速度
        - 随机微小偏移
        """
        if duration_ms is None:
            duration_ms = random.randint(300, 800)
        
        points = []
        steps = duration_ms // 16  # 60fps
        
        # 控制点 (用于贝塞尔曲线)
        cp1 = {
            'x': start['x'] + (end['x'] - start['x']) * 0.3 + random.uniform(-50, 50),
            'y': start['y'] + (end['y'] - start['y']) * 0.3 + random.uniform(-50, 50)
        }
        cp2 = {
            'x': start['x'] + (end['x'] - start['x']) * 0.7 + random.uniform(-50, 50),
            'y': start['y'] + (end['y'] - start['y']) * 0.7 + random.uniform(-50, 50)
        }
        
        for i in range(steps + 1):
            t = i / steps
            
            # 三阶贝塞尔曲线
            x = (
                (1-t)**3 * start['x'] +
                3 * (1-t)**2 * t * cp1['x'] +
                3 * (1-t) * t**2 * cp2['x'] +
                t**3 * end['x']
            )
            y = (
                (1-t)**3 * start['y'] +
                3 * (1-t)**2 * t * cp1['y'] +
                3 * (1-t) * t**2 * cp2['y'] +
                t**3 * end['y']
            )
            
            # 添加微小随机偏移 (模拟手抖)
            x += random.uniform(-2, 2)
            y += random.uniform(-2, 2)
            
            points.append({'x': x, 'y': y, 't': t * duration_ms})

                    "bbox": [x, y, width, height],
                    "purpose": "元素用途描述",
                    "confidence": 0.0-1.0
                }}
            ],
            "page_description": "页面整体描述",
            "suggested_action": "建议的下一步操作"
        }}
        """
        
        response = requests.post(
            f"{self.ollama_url}/api/generate",
            json={
                "model": self.model,
                "prompt": prompt,
                "images": [image_b64],
                "stream": False
            }
        )
        
        result = response.json()
        return self._parse_response(result["response"])
    
    async def identify_element_by_description(
        self,
        screenshot: bytes,
        description: str
    ) -> Optional[dict]:
        """
        通过文字描述识别元素
        
        Args:
            screenshot: 页面截图
            description: 元素描述 (如"蓝色的提交按钮，在表单底部")
        
        Returns:
            {
                "found": True,
                "bbox": [x, y, width, height],
                "confidence": 0.92
            }
        """
        result = await self.analyze_page(
            screenshot,
            f"请找到：{description}"
        )
        
        if result.get("elements"):
            best_match = max(
                result["elements"],
                key=lambda e: e.get("confidence", 0)
            )
            return {
                "found": True,
                "bbox": best_match["bbox"],
                "confidence": best_match["confidence"]
            }
        
        return {"found": False}
    
    def _parse_response(self, response_text: str) -> dict:
        """解析 VLM 响应"""
        import json
        import re
        
        # 提取 JSON 部分
        json_match = re.search(r'\{.*\}', response_text, re.DOTALL)
        if json_match:
            try:
                return json.loads(json_match.group())
            except json.JSONDecodeError:
                pass
        
        return {
            "elements": [],
            "page_description": response_text,
            "suggested_action": ""
        }
```

router = APIRouter()

    ai_service: AIService = Depends(get_ai_service),
    current_user: User = Depends(get_current_user)
):

    AI 生成自动化流程

        elements=result["elements"],

## 6. 安全设计

### 6.1 凭证加密方案

```python
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64
import os

class CredentialEncryption:
    """凭证加密服务"""
    
    def __init__(self, master_key: bytes):
        self.master_key = master_key
        self.salt = os.urandom(16)
        self.fernet = self._derive_key()
    
    def _derive_key(self) -> Fernet:
        """从主密钥派生加密密钥"""
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=self.salt,
            iterations=100000,
        )
        key = base64.urlsafe_b64encode(kdf.derive(self.master_key))
        return Fernet(key)
    
    def encrypt(self, plaintext: str) -> str:
        """加密凭证"""
        encrypted = self.fernet.encrypt(plaintext.encode())
        return base64.b64encode(encrypted).decode()
    
    def decrypt(self, ciphertext: str) -> str:
        """解密凭证"""
        encrypted = base64.b64decode(ciphertext.encode())
        decrypted = self.fernet.decrypt(encrypted)
        return decrypted.decode()
```
